2) CLI -- Prepare A Config Map with some environment variables : Inject environment variables into a workbench from a Config Map : 
[developer]$  oc project
[developer]$ oc create cm db-config \
   --from-literal=db_host=postgresl \
   --from-literal=db_pass=pgl_pass \
   --from-literal=db_user=developer
   
[developer]$ $ oc get cm db-config -o yaml > db_configmap.yaml
[developer]$ vi db_configmap.yaml
[developer]$  cat db_configmap.yaml
