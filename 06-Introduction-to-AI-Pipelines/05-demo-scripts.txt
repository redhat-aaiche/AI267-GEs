pipeline.py

from kfp import dsl
from kfp import compiler

@dsl.component
def say_hello(name: str) -> str:
    hello_text = f'Hello, {name}!'
    print(hello_text)
    return hello_text

@dsl.pipeline
def hello_pipeline(recipient: str) -> str:
    hello_task = say_hello(name=recipient)
    return hello_task.output

compiler.Compiler().compile(hello_pipeline, 'pipeline.yaml')



(.venv) $ python pipeline.py
(.venv) $ ls pipeline.*

from kfp import dsl
from kfp import compiler
from kfp.compiler.compiler_utils import KubernetesManifestOptions


@dsl.component
def say_hello(name: str) -> str:
    hello_text = f'Hello, {name}!'
    print(hello_text)
    return hello_text

@dsl.pipeline
def hello_pipeline(recipient: str) -> str:
    hello_task = say_hello(name=recipient)
    return hello_task.output

if __name__ == "__main__":
    outfile = "pipeline-manifest.yaml"
    # Compile the pipeline
    compiler.Compiler().compile(
        hello_pipeline,
        outfile,
        kubernetes_manifest_format=True,
        kubernetes_manifest_options=KubernetesManifestOptions(
            pipeline_name="my-second-pipeline",
            pipeline_display_name="My Second Pipeline",
            pipeline_version_name="my-second-pipeline-v1",
            pipeline_version_display_name="My Second Pipeline v1",
            namespace="dspa-kubeflow",
            include_pipeline_manifest=True
        )
    )

    print(
        "Pipeline compiled.\n"
        f"Use the Kubernetes API to import the '{outfile}' manifest file"
    )
    
    
(.venv) $ python pipeline-manifest.py
(.venv) $ ls pipeline-manifest.*
